class Solution {
    public int maxTwoEvents(int[][] events) {
        // sorting according to the start time
        Arrays.sort(events, (a, b) -> a[0] - b[0]);
        int[] suffix = new int[events.length];
        int max = 0;
        for(int i = events.length - 1; i >= 0; i--) {
            max = Math.max(max, events[i][2]);
            suffix[i] = max;
        }

        int ans = 0;
        for(int i = 0; i < events.length; i++) {
            int[] cur = events[i];
            ans = Math.max(ans, cur[2]);
            int l = i + 1, r = events.length - 1;
            while(l < r) {
                int mid = (l + r) / 2;
                if(events[mid][0] <= cur[1]) {
                    l = mid + 1;
                } else {
                    r = mid;
                }
            }
            if(l < events.length && events[l][0] > cur[1]) {
                ans = Math.max(ans, cur[2] + suffix[l]);
            }
        }
        return ans;
    }
}
